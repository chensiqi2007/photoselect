cmake_minimum_required(VERSION 3.16)
project(photoselect)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED OHOS_ARCH)
  set(OHOS_ARCH arm64-v8a)
endif()
add_library(photoselect_core STATIC IMPORTED)
set_target_properties(photoselect_core PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/prebuilt/${OHOS_ARCH}/libphotoselect_harmony_core.a")
add_library(photoselect SHARED napi_init.cpp)
target_link_libraries(photoselect PRIVATE photoselect_core libace_napi.z.so)
